# 이번 실습에서 자주 쓰는 명령어

```
## 상태확인:get
kubectl get nodes
kubectl get deploy,po,svc -o wide # 길게 로그 뽑음

## 상세 확인:describe
kubectl describe pod <pod-name> # <>꺽새는 쓰는 거 아님 진짜 파드 네임 넣음
kubectl describe svc <svc-name>

## 로그:log
kubectl logs <pod-name>

## 테스트:curl
kubectl exec -it <pod-name> -- curl localhost

## 정리:delete
kubectl delete -f 10.k8s-basics/10.ninx-nodeport.yaml
```
